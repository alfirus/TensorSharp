cmake_minimum_required(VERSION 3.20)
project(TensorSharpCpu LANGUAGES CXX)
add_library(TensorSharpCpu SHARED embedding_q16_vnni.cpp embedding_attention_avx512.cpp)
target_compile_features(TensorSharpCpu PRIVATE cxx_std_17)
set_target_properties(TensorSharpCpu PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(TensorSharpCpu PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Werror)
endif()
install(TARGETS TensorSharpCpu LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
